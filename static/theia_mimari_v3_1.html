# gatekeeper.py — v2.3 · Risk Sınıflandırıcı + Onay Akışı
# Güvenlik mimaride yaşar, prompt'ta değil

import re, logging, asyncio
from enum import Enum
from datetime import datetime

# ── Risk Seviyeleri ──────────────────────────────────────────
class RiskLevel(Enum):
    LOW      = 1   # Normal mesaj → direkt geç
    MEDIUM   = 2   # Şüpheli ama muhtemelen iyi niyet → logla, geç
    HIGH     = 3   # Onay iste → Telegram'a yönlendir
    CRITICAL = 4   # Blokla → kesinlikle geçirme

# ── Injection Tespit Örüntüleri ──────────────────────────────
INJECTION_PATTERNS = [
    r"ignore previous instructions",
    r"ignore all instructions",
    r"you are now (a|an|the)",
    r"new persona",
    r"pretend (you are|to be)",
    r"jailbreak",
    r"DAN mode",
    r"system prompt",  # hassas bağlamlarda
    r"</?system>",
    r"\[INST\]",
]

# ── Risk Sınıflandırıcı ──────────────────────────────────────
def classify_risk(message: str) -> RiskLevel:
    text = message.lower()

    # CRITICAL — injection tespit
    for pattern in INJECTION_PATTERNS:
        if re.search(pattern, text, re.IGNORECASE):
            audit_log("CRITICAL", message, pattern)
            return RiskLevel.CRITICAL

    # HIGH — tehlikeli içerik kategorileri
    high_keywords = ["bomba", "silah", "patlayıcı", "hack", "saldır"]
    if any(kw in text for kw in high_keywords):
        audit_log("HIGH", message)
        return RiskLevel.HIGH

    # MEDIUM — şüpheli ama işlenebilir
    medium_keywords = ["şifre", "token", "api key"]
    if any(kw in text for kw in medium_keywords):
        audit_log("MEDIUM", message)
        return RiskLevel.MEDIUM

    return RiskLevel.LOW

# ── Telegram Onay Akışı (HIGH) ───────────────────────────────
async def request_approval(bot, chat_id: int, message: str) -> bool:
    """HIGH risk mesaj için Kaptan'dan Telegram onayı iste."""
    approval_msg = await bot.send_message(
        chat_id,
        f"⚠️ Yüksek risk tespit edildi:\n{message[:200]}...\n\nOnaylıyor musun?",
        reply_markup=APPROVAL_KEYBOARD  # Evet/Hayır inline keyboard
    )
    # CallbackQuery bekle (timeout: 60s)
    result = await wait_for_approval(approval_msg.message_id, timeout=60)
    return result

# ── Denetim Logu ─────────────────────────────────────────────
def audit_log(level: str, message: str, pattern: str = ""):
    logging.getLogger("gatekeeper").warning(
        f"[{datetime.now().isoformat()}] RISK={level} | pattern={pattern!r} | msg={message[:100]!r}"
    )

# ── Ana Giriş Noktası ────────────────────────────────────────
async def gate(bot, chat_id: int, message: str) -> tuple[bool, RiskLevel]:
    """True döndürürse mesaj LLM'e geçer."""
    risk = classify_risk(message)

    if risk == RiskLevel.CRITICAL:  return False, risk
    if risk == RiskLevel.HIGH:      return await request_approval(bot, chat_id, message), risk
    return True, risk  # LOW/MEDIUM → geç
# soul_agent.py — Sabit Zemin Bağlamı · 5dk Cache
# Her mesajda core_identity + daily_summary çeker

import aiohttp, time
from config import SOUL_API_URL

# ── In-Memory Cache ──────────────────────────────────────────
_cache = {
    "data": None,
    "timestamp": 0,
    "ttl": 300  # 5 dakika (saniye)
}

# ── Ana Fonksiyon ────────────────────────────────────────────
async def get_soul_context() -> str:
    """Soul API'den core_identity + güncel daily_summary çek.
    Cache geçerliyse API'ye gitme. Hata durumunda boş string dön."""

    # Cache kontrolü
    if _cache["data"] and (time.time() - _cache["timestamp"]) < _cache["ttl"]:
        return _cache["data"]

    try:
        # Soul API'ye GET isteği
        async with aiohttp.ClientSession() as session:
            async with session.get(
                f"{SOUL_API_URL}/api/memory",
                timeout=aiohttp.ClientTimeout(total=5)
            ) as resp:
                if resp.status != 200:
                    return ""  # graceful fallback
                data = await resp.json()

        # core_identity + en güncel daily_summary birleştir
        core = data.get("core_identity", "")
        daily = _get_latest_daily(data.get("entries", []))

        result = _format_soul_context(core, daily)

        # Cache güncelle
        _cache["data"] = result
        _cache["timestamp"] = time.time()
        return result

    except Exception as e:
        # Soul down veya timeout → boş string, bot çalışmaya devam eder
        return ""

# ── Yardımcı Fonksiyonlar ────────────────────────────────────
def _get_latest_daily(entries: list) -> str:
    """entry_type=daily_summary içinden en güncel olanı seç."""
    dailies = [e for e in entries if e.get("entry_type") == "daily_summary"]
    if not dailies: return ""
    latest = max(dailies, key=lambda x: x.get("updated_at", ""))
    return latest.get("content", "")

def _format_soul_context(core: str, daily: str) -> str:
    parts = []
    if core:  parts.append(f"[KİMLİK]\n{core}")
    if daily: parts.append(f"[BUGÜN]\n{daily}")
    return "\n\n".join(parts)

def invalidate_cache():
    """Bridge yeni veri push'ladığında cache'i sıfırla."""
    _cache["timestamp"] = 0
# obsidian_bridge.py — Obsidian → Soul DB Köprüsü
# cron: */15 * * * * — 15dk'da bir çalışır

import os, requests, logging
from pathlib import Path
from datetime import datetime

VAULT_SYSTEM_DIR = Path("~/TheiaMemory/System").expanduser()
SOUL_API_URL     = "http://127.0.0.1:8000"
LOG_FILE         = Path("~/theia-soul/bridge.log").expanduser()

logging.basicConfig(
    filename=LOG_FILE, level=logging.INFO,
    format="%(asctime)s [%(levelname)s] %(message)s"
)

# ── Ana Köprü Fonksiyonu ─────────────────────────────────────
def run_bridge():
    logging.info("Bridge başladı")
    synced, errors = 0, 0

    # Vault System/ klasöründeki tüm .md dosyalarını gez
    for md_file in VAULT_SYSTEM_DIR.glob("*.md"):
        try:
            content = md_file.read_text(encoding="utf-8")
            key     = md_file.stem  # dosya adı = memory key

            # entry_type belirle (core_identity özel)
            entry_type = "core" if key == "core_identity" else "memory"

            # Soul API'ye PUT
            resp = requests.put(
                f"{SOUL_API_URL}/api/memory/{key}",
                json={"content": content, "entry_type": entry_type},
                timeout=10
            )
            resp.raise_for_status()
            synced += 1
            logging.info(f"✓ {key} → Soul DB")

        except Exception as e:
            errors += 1
            logging.error(f"✗ {md_file.name}: {e}")

    # Scout decay kontrolünü tetikle
    trigger_scout()

    logging.info(f"Bridge bitti — {synced} sync, {errors} hata")

# ── Scout Tetikleyici ────────────────────────────────────────
def trigger_scout():
    """Bridge her çalıştığında Scout decay kontrolü tetiklenir."""
    try:
        resp = requests.post(
            f"{SOUL_API_URL}/api/scout/check",
            timeout=10
        )
        summary = resp.json()
        logging.info(
            f"Scout: {summary.get('passived', 0)} passive, {summary.get('archived', 0)} archived"
        )
    except Exception as e:
        logging.warning(f"Scout tetiklenemedi: {e}")

if __name__ == "__main__":
    run_bridge()
# handlers/message.py — Ana Mesaj İşleyici
# soul_ctx + memory_ctx + web_ctx → gatekeeper → Claude

import asyncio
from agents.soul_agent   import get_soul_context
from agents.memory_agent import get_context
from agents.web_agent    import get_web_context
from gatekeeper          import gate, RiskLevel
from llm_client          import call_claude

# ── Ana Giriş Noktası ────────────────────────────────────────
async def handle_message(update, context):
    user_msg = update.message.text
    chat_id  = update.effective_chat.id

    # 1. GATEKEEPER — LLM'e gitmeden önce risk kontrolü
    allowed, risk_level = await gate(context.bot, chat_id, user_msg)
    if not allowed:
        await update.message.reply_text("⛔ Bu isteği işleyemiyorum.")
        return

    # 2. PARALEL BAĞLAM TOPLAMA
    soul_ctx, memory_ctx = await asyncio.gather(
        get_soul_context(),          # core_identity + daily_summary
        get_context(user_msg),        # FTS vault arama
        return_exceptions=True
    )
    # Exception gelirse boş string → graceful
    soul_ctx   = soul_ctx   if isinstance(soul_ctx,   str) else ""
    memory_ctx = memory_ctx if isinstance(memory_ctx, str) else ""

    # Web agent sadece 🌍 prefix ile tetiklenir
    web_ctx = ""
    if user_msg.startswith("🌍"):
        web_ctx = await get_web_context(user_msg[1:].strip())

    # 3. SİSTEM PROMPT İNŞASI
    system_prompt = _build_system(soul_ctx, memory_ctx, web_ctx)

    # 4. CLAUDE API ÇAĞRISI
    response_text = await call_claude(
        system=system_prompt,
        user=user_msg,
        history=_get_history(chat_id)
    )

    # 5. YANIT GÖNDER + GEÇMİŞE KAYDET
    await update.message.reply_text(response_text)
    _save_history(chat_id, user_msg, response_text)
