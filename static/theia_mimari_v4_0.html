class RiskLevel(Enum):
    LOW      = 1   # Normal mesaj → geç
    MEDIUM   = 2   # Şüpheli → logla, geç
    HIGH     = 3   # Telegram onayı iste
    CRITICAL = 4   # Blokla

INJECTION_PATTERNS = [
    r"ignore previous instructions",
    r"you are now (a|an|the)",
    r"jailbreak",
    r"DAN mode",
    r"system prompt",
]

async def gate(bot, chat_id, message) -> tuple[bool, RiskLevel]:
    risk = classify_risk(message)
    if risk == RiskLevel.CRITICAL: return False, risk
    if risk == RiskLevel.HIGH:     return await request_approval(...), risk
    return True, risk  # LOW/MEDIUM → geç
# api/routes.py — chat() · God Node #4 · betweenness 0.058
# API Routes ↔ Model Layer ↔ DB arasında köprü

@router.post("/chat", response_model=ChatResponse)
async def chat(req: ChatRequest):
    model = factory.get_model(req.model)       # model fabrikası
    history = await db.get_messages(req.session_id)
    if not history:
        await db.create_session(req.session_id, model.model_id)

    messages = [Message(role=m["role"], content=m["content"]) for m in history]
    messages.append(Message(role="user", content=req.message))

    gen_req = GenerateRequest(               # God Node #10 · betweenness 0.051
        messages=messages,
        system=await _build_system(req.system),
        max_tokens=req.max_tokens,
    )
    response = await model.generate(gen_req)

    await db.add_message(req.session_id, "user", req.message)
    await db.add_message(req.session_id, "assistant", response.content)
    return ChatResponse(...)

async def _build_system(req_system):
    if req_system is not None: return req_system  # override
    entries = await db.list_memory()
    user_memory = "\n".join(f"{e['key']}: {e['value']}" for e in entries)
    return _theia_system.build_system(user_memory=user_memory)

_WEB_SUFFIX = """
## Web Arama Sonuçları

def build_system(*, web=False, vault_context="", web_context="", user_memory="") -> str:
    now = datetime.now().strftime("%Y-%m-%d %H:%M")
    text = _BASE + (_WEB_SUFFIX if web else "")
    text += f"\n\nŞu anki tarih ve saat: {now} (UTC+3)"
    if user_memory:
        text += f"\n\nKullanıcı hafızası:\n{user_memory}"

# models/factory.py — Çoklu Model Fabrikası
# Graphify: _build() god node #6 · 12 edge

_registry: dict[str, BaseModel] = {}  # singleton cache

def get_model(name: str | None = None) -> BaseModel:
    key = (name or config.DEFAULT_MODEL).lower()
    if key not in _registry:
        _registry[key] = _build(key)
    return _registry[key]

def _build(key: str) -> BaseModel:
    if key == "claude":    return ClaudeModel(model_id=config.CLAUDE_MODEL_ID, ...)
    if key == "deepseek": return DeepSeekModel(model_id=config.DEEPSEEK_MODEL_ID, ...)
    if key == "kimi":     return KimiModel(model_id=config.KIMI_MODEL_ID, ...)
    if key == "ollama":   return OllamaModel(model_id=config.OLLAMA_MODEL_ID, ...)
    raise ValueError(f"Bilinmeyen model: {key}")
